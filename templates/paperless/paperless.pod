[Pod]
PodName=paperless
# beyond loopback so Haus's Haushalt reaches the API from its container (haus#11, ADR 0012 §4);
# ports lan: false keeps the LAN out, and the API wants a token
PublishPort=8000:8000
# the OIDC discovery and token calls go to Authelia through Caddy on this box (a
# container in a pod cannot set its own hosts); the host gateway, not the LAN IP,
# which under pasta stays inside the pod (#252)
AddHost=auth.${DOMAIN}:host-gateway
